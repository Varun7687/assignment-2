  reporting_user_data = <<-USERDATA
	#!/bin/bash
	set -euxo pipefail
	
	PROJECT_DIR="/home/ec2-user/northwind-reporting"
	SCRIPT_PATH="$PROJECT_DIR/generate_report.sh"
	TEMPLATE_PATH="$PROJECT_DIR/report.html.template"
	LOG_PATH="$PROJECT_DIR/generate_report.log"
	
	mkdir -p "$PROJECT_DIR"
	cat > "$TEMPLATE_PATH" <<'REPORT_HTML_TEMPLATE'
	${local.report_template}
	REPORT_HTML_TEMPLATE
	
	cat > "$SCRIPT_PATH" <<'REPORT_SCRIPT'
	#!/usr/bin/env bash
	set -euo pipefail
	umask 027
	
	export PATH="/usr/local/bin:/usr/bin:/bin"
	AWS_REGION="${var.aws_region}"
	REPORT_BUCKET="${aws_s3_bucket.reports.bucket}"
	SNS_TOPIC_ARN="${aws_sns_topic.report_ready.arn}"
	TEMPLATE_PATH="/home/ec2-user/northwind-reporting/report.html.template"
	
	RUN_ID="$(date -u +'%Y-%m-%dT%H-%M-%S')-$$"
	REPORT_KEY="reports/$${RUN_ID}.html"
	REPORT_FILE="$(mktemp /tmp/northwind-report.XXXXXX.html)"
	trap 'rm -f "$REPORT_FILE"' EXIT
	
	REPORT_DATE="$(date -u +'%d %B %Y')"
	REPORT_HOST="$(hostname)"
	DISK_USAGE="$(df -h / | awk 'NR==2 {print $5 " used of " $2}')"
	
	sed \
	  -e "s|__REPORT_DATE__|$REPORT_DATE|g" \
	  -e "s|__REPORT_HOST__|$REPORT_HOST|g" \
	  -e "s|__DISK_USAGE__|$DISK_USAGE|g" \
	  -e "s|__REPORT_KEY__|$REPORT_KEY|g" \
	  "$TEMPLATE_PATH" > "$REPORT_FILE"
	
	aws s3 cp "$REPORT_FILE" "s3://$${REPORT_BUCKET}/$${REPORT_KEY}" \
	  --region "$AWS_REGION" \
	  --sse AES256 \
	  --content-type "text/html" \
	  --only-show-errors
	
	PRESIGNED_URL="$(aws s3 presign "s3://$${REPORT_BUCKET}/$${REPORT_KEY}" \
	  --expires-in 3600 \
	  --region "$AWS_REGION")"
	
	printf -v SNS_MESSAGE 'Report uploaded to s3://%s/%s.\n\nDownload link valid for approximately 1 hour:\n%s' \
	  "$REPORT_BUCKET" "$REPORT_KEY" "$PRESIGNED_URL"
	
	aws sns publish \
	  --topic-arn "$SNS_TOPIC_ARN" \
	  --subject "Northwind report ready" \
	  --message "$SNS_MESSAGE" \
	  --region "$AWS_REGION" \
	  >/dev/null
	
	printf 'Uploaded HTML report: s3://%s/%s\n' "$REPORT_BUCKET" "$REPORT_KEY"
	printf 'SNS notification sent with a temporary download link.\n'
	REPORT_SCRIPT
	
	chmod 750 "$SCRIPT_PATH"
	chmod 640 "$TEMPLATE_PATH"
	chown -R ec2-user:ec2-user "$PROJECT_DIR"
	
	dnf install -y cronie
	systemctl enable --now crond
	
	cat > /etc/cron.d/northwind-reporting <<CRON
	SHELL=/bin/bash
	PATH=/usr/local/bin:/usr/bin:/bin
	${var.cron_schedule} ec2-user AWS_REGION=${var.aws_region} REPORT_BUCKET=${aws_s3_bucket.reports.bucket} SNS_TOPIC_ARN=${aws_sns_topic.report_ready.arn} $${SCRIPT_PATH} >> $${LOG_PATH} 2>&1
	CRON
	chmod 644 /etc/cron.d/northwind-reporting
	USERDATA
}
